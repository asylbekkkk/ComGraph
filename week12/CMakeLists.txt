cmake_minimum_required(VERSION 3.10)
project(week12 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(glfw3 REQUIRED)
find_package(OpenGL REQUIRED)
add_executable(week12 main.cpp ../external/glad/src/gl.c)
target_include_directories(week12 PRIVATE ../external/glad/include ../external/stb)
target_compile_definitions(week12 PRIVATE ASSET_DIR="${CMAKE_CURRENT_SOURCE_DIR}/assets")
target_link_libraries(week12 PRIVATE glfw OpenGL::GL ${CMAKE_DL_LIBS})
